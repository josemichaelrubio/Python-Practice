Two-sum / complement
def two_sum(nums, target):
    seen = {}           # val → index
    for i, n in enumerate(nums):
        comp = target - n
        if comp in seen:
            return [seen[comp], i]
        seen[n] = i
    return []
O(n) time O(n) space naive O(n²)
Sliding window — fixed
# max sum of k-length subarray
def max_sum_k(arr, k):
    window = sum(arr[:k])
    best = window
    for i in range(k, len(arr)):
        window += arr[i] - arr[i-k]
        best = max(best, window)
    return best
O(n) time naive O(n·k)
Sliding window — variable
# longest substring with no repeats
def length_of_longest(s):
    seen = {}
    left = best = 0
    for right, ch in enumerate(s):
        if ch in seen and seen[ch] >= left:
            left = seen[ch] + 1
        seen[ch] = right
        best = max(best, right - left + 1)
    return best

# sliding window anagram (fixed k)
def find_anagrams(s, p):
    need = Counter(p)
    window = Counter(s[:len(p)])
    res = [0] if window == need else []
    for i in range(len(p), len(s)):
        window[s[i]] += 1
        old = s[i - len(p)]
        window[old] -= 1
        if window[old] == 0:
            del window[old]
        if window == need:
            res.append(i - len(p) + 1)
    return res
Majority element / Boyer-Moore
# element appearing > n/2 times
def majority(nums):
    candidate, count = None, 0
    for n in nums:
        if count == 0:
            candidate = n
        count += 1 if n == candidate else -1
    return candidate
O(n) time O(1) space

String concatenation in a loop
The trap: strings are immutable — s += c in a loop is O(n²) because it creates a new string each iteration.
# BAD — O(n²)
result = ""
for ch in chars:
    result += ch

# GOOD — O(n)
result = []
for ch in chars:
    result.append(ch)
return "".join(result)

Modifying dict while iterating
The trap: for k in d: del d[k] raises RuntimeError. Collect keys first.
# BAD
for k in freq:
    if freq[k] == 0:
        del freq[k]   # RuntimeError!

# GOOD
to_delete = [k for k, v in freq.items() if v == 0]
for k in to_delete:
    del freq[k]

# or use dict comprehension
freq = {k: v for k, v in freq.items() if v > 0}
